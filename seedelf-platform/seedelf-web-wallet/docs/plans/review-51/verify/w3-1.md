# w3#1: verification

From lane W3, graded low by its finder.

## Trace: confirmed, low

Holds as described. A Mix from the Lovejoin tile is reviewed and funded at the Settings depth of that moment. Its record keeps no depth: mixOutSubmit writes `mix: { boxes, again?, publicToo?, bootstrap?, noCollateral?, pool?, width? }` and no `auto.lovejoin`. So buildBack passes `fixed = undefined`, and chain() and plan() read Settings' depth again at build time. The funded ADA is then re-counted at the new depth, capped at the funded box count.

I found no guard, no other writer of the depth, and no test that pins the current behaviour. Width is kept for exactly this reason (`mix.width`, chunk 34), and a swap's depth is kept too (L21). Depth is the missing twin. My Vitest fails on TREE: 2 boxes reviewed at 1 wave became 1 box at 2 waves with 4 mixes. With the finder's 3-line fix applied in a copy, the test passes.

Mix my boxes again takes the same record and the same buildBack/chain path. I traced it but didn't run it.

Graded low:
- **Who:** only the user can reach it, by changing Settings' depth in the window between Send and the runner's build. That window is the funding landing plus at most a minute's alarm, or longer if the wallet locks first: until the next unlock and its drawn wait.
- **Cost:** nothing is lost or stranded. The extra mix fees buy mixing at the depth the user last chose; the boxes that no longer fit come back at once. In the other direction, a box hides among fewer leaves than the review said, and the fee money comes back.
- **What's wrong:** the review's figures (boxes, mixes, fees, how many leaves a box hides among) don't hold for that run, and nothing says so. If the pool is too small for the new depth, the whole Mix comes back unmixed, with a skip reason.
- **Precedent:** the same class for swaps was graded Low (independent review L21).

// Chunk 51, w3#1: a mix from the Lovejoin tile is built at the depth its review showed and its funding paid for,
// whatever Settings' depth says by the time its funding lands and the runner builds its chain: as its width is kept
// (`mix.width`, chunk 34's 34e review, 4) and a swap's depth is (independent review L21). With the real WebAssembly
// and Lovejoin v2's 5 ₳ pool, as lovejoin-planner.test.ts's width twin ("its chain goes four wide whatever Settings
// says since") runs it.
import { describe, expect, it, vi } from "vitest";

import { Collateral } from "../src/background/collateral";
import type { KoiosUtxo } from "../src/background/koios";
import { LovejoinService, type LovejoinChain, type PoolRef } from "../src/background/lovejoin";
import { builtOutputs, Minswap } from "../src/background/minswap";
import { SessionService } from "../src/background/sessions";
import { txIdOf } from "./fixtures/cbor";
import { loadTestWasm, sessionSwap, testBalances, vectors } from "./fakes";
import { ADA, ADA_POOL, adaStrangers } from "./v2-pools";

const PASSWORD = "correct horse battery";
const PHRASE = (vectors("cardano_account.json") as Array<{ account: number; phrase: string }>).find(
  (v) => v.account === 0 && v.phrase.split(" ").length === 12,
)!.phrase;

/** Ogmios's answer when the network measures no more than a transaction declares. */
const AGREES = { jsonrpc: "2.0", method: "evaluateTransaction", result: [] };

/** Real chains are built and measured in WebAssembly: seconds each. */
const SLOW = { timeout: 180_000 };

type T = ReturnType<typeof testBalances>;

async function wallet(): Promise<T> {
  const t = testBalances();
  await t.wallet.create(PHRASE, PASSWORD);
  t.koios.evaluation = AGREES;
  // A slot in preprod's Shelley era, for what the account signs (lovejoin-flows.test.ts).
  t.koios.tip = 110_000_000;
  t.collateral.answer = { status: 200, body: { witness: "a1008182" } };
  return t;
}

/** Lovejoin reading `pools`, giveme.my's witness stood in for. */
function pooled(t: T, pools: PoolRef[]): LovejoinService {
  const wasm = loadTestWasm();
  return new LovejoinService({
    ...t.deps,
    wasm: {
      ...wasm,
      finishLovejoinWithdraw: (request: string) => {
        const { txCbor } = JSON.parse(request) as { txCbor: string };
        return JSON.stringify({ txCbor, txHash: txIdOf(Uint8Array.from(Buffer.from(txCbor, "hex"))) });
      },
    } as typeof wasm,
    collateral: () => new Collateral("https://www.giveme.my/preprod/collateral/", t.collateral.fetch),
    store: t.store,
    pools: async () => pools,
    random: () => 0.5,
  });
}

/** The sessions over `lj`, giveme.my and the one-time key's signatures stood in for. */
function runner(t: T, lj: LovejoinService): SessionService {
  const wasm = loadTestWasm();
  return new SessionService({
    ...t.deps,
    wasm: {
      ...wasm,
      signScriptSpend: (_key: unknown, request: string) => {
        const { txCbor } = JSON.parse(request) as { txCbor: string };
        return JSON.stringify({ txCbor, txHash: txIdOf(Uint8Array.from(Buffer.from(txCbor, "hex"))) });
      },
    } as typeof wasm,
    collateral: () => new Collateral("https://www.giveme.my/preprod/collateral/", t.collateral.fetch),
    store: t.store,
    minswap: () => new Minswap("https://aggr.monorepo-testnet-preprod.minswap.org/aggregator", t.minswap.fetch),
    lovejoin: lj,
    sleep: async () => undefined,
  });
}

/** A UTxO at session 0's account, as Koios lists it. */
function atSession(tx_hash: string, tx_index: number, value: string): KoiosUtxo {
  return {
    tx_hash,
    tx_index,
    address: sessionSwap.address,
    value,
    stake_address: null,
    payment_cred: sessionSwap.keyHash,
    epoch_no: 315,
    block_height: 5_000_000,
    block_time: 1_800_000_000,
    datum_hash: null,
    inline_datum: null,
    reference_script: null,
    is_spent: false,
    asset_list: [],
  } as KoiosUtxo;
}

/** What funding `id` (submitted) paid session 0's account, read from the transaction itself, as Koios lists it once it lands. */
function landed(t: T, id: string): KoiosUtxo[] {
  const bytes = t.koios.submitted.find((b) => txIdOf(b) === id);
  if (!bytes) throw new Error(`not submitted: ${id}`);
  return builtOutputs(bytes).flatMap((o, i) => (o.address.slice(2, 58) === sessionSwap.keyHash ? [atSession(id, i, o.lovelace.toString())] : []));
}

describe("a tile Mix's chain keeps the depth its review showed (chunk 51, w3#1)", SLOW, () => {
  it("builds the reviewed boxes at the reviewed depth, though Settings' depth changed between Send and the build", async () => {
    const t = await wallet();
    await t.deps.preferences.set({ lovejoinDepth: 1 });
    const lj = pooled(t, [ADA_POOL]);
    const sessions = runner(t, lj);
    // Twenty strangers' boxes in the 5 ₳ pool: room for two boxes at either depth, and its collateral's mix.
    t.koios.addedToAccounts.push(...adaStrangers(20, 0x100, t.clock.now));

    // The review: two boxes of 5 ₳, one wave deep, two mixes, funded for exactly that (2 × 5.95 ₳ + 1.5 ₳).
    const out = await sessions.mixOutBuild("preprod", 2, { pool: ADA });
    expect(out.mix).toMatchObject({ boxes: 2, depth: 1, mixes: 2, lovelace: (2 * 5_950_000 + 1_500_000).toString() });
    const { index } = await sessions.mixOutSubmit("preprod", out.txHash);

    // Sent. Before the funding lands, the user sets Settings' depth to two waves.
    await t.deps.preferences.set({ lovejoinDepth: 2 });

    // The funding lands as it paid the account, and the runner builds the session's chain.
    const paid = landed(t, out.txHash);
    expect(paid).toHaveLength(2);
    t.koios.addedToAccounts.push(...paid);
    t.koios.confirmations = 1;
    const chain = vi.spyOn(lj, "chain");
    await sessions.advance("preprod", index, true);
    expect(chain).toHaveBeenCalledTimes(1);
    const built = (await chain.mock.results[0]!.value) as LovejoinChain | undefined;
    expect(built, "the runner built no chain for the mix").toBeDefined();

    // What it built, against what its review showed and its funding paid for.
    const ran = {
      depth: built!.depth,
      boxes: built!.boxes,
      mixes: built!.txs.filter((x) => x.kind === "mix" && !x.tail).length,
    };
    expect(ran).toEqual({ depth: out.mix.depth, boxes: out.mix.boxes, mixes: out.mix.mixes });
  });
});
