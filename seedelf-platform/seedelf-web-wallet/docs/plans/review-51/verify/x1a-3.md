// Chunk 51, x1a#3: a site's transaction with a CIP-20 note and other metadata beside it. The body commits to all of
// it (its auxiliary data hash), so signing publishes all of it, under the account's signature, for good. The sign
// window says "It carries metadata, which anyone can read." only when there's no note (DappApprovals.tsx:1083): with
// one, it shows the note's lines and names nothing else, though WebAssembly's summary says there is metadata. Read
// through the real WebAssembly (inspectDappTx, the worker's readTx call; decodeTx, Transaction details' reading) and
// rendered as the window renders it.
import { readFileSync } from "node:fs";
import { blake2b } from "@noble/hashes/blake2.js";
import { bytesToHex } from "@noble/hashes/utils.js";
import { createElement } from "react";
import { renderToStaticMarkup } from "react-dom/server";
import { describe, expect, it } from "vitest";

import type { DappTxSummary, TxDetail, TxMetadatum } from "../src/shared/rpc";
import { NetworkContext } from "../src/ui/network";
import { SignTx } from "../src/ui/screens/DappApprovals";
import { loadTestWasm, vectors } from "./fakes";

const wasm = loadTestWasm();
const account = (words: number) =>
  vectors("cardano_account.json").find((v) => v.account === 0 && v.phrase.split(" ").length === words)!;
const ME = account(12);
const OWN = ME.preprod.receive_0 as string;
const SITE = account(15).preprod.receive_0 as string;
const INPUT = "11".repeat(32);

const utf8 = (s: string) => Buffer.from(s, "utf8").toString("hex");
/** A CBOR text string of fewer than 256 bytes. */
const text = (s: string) => {
  const n = Buffer.byteLength(s, "utf8");
  return `${n < 24 ? (0x60 + n).toString(16) : `78${n.toString(16).padStart(2, "0")}`}${utf8(s)}`;
};
const bytesOf = (hex: string) => `58${(hex.length / 2).toString(16).padStart(2, "0")}${hex}`;

const NOTE = "Thanks for order 77";
/** CIP-20's message: 674 → {"msg": [NOTE]}. */
const note = `1902a2a1${text("msg")}81${text(NOTE)}`;
/** What else the site writes, under a label of its own: 1337 → {"email": "alice@example.com", "order": 77}. */
const other = `190539a2${text("email")}${text("alice@example.com")}${text("order")}184d`;

/**
 * A site's transaction: the account's 10 ₳ UTxO pays the site 4 ₳, with `metadata` (a CBOR map, hex) as its auxiliary
 * data and that data's hash in the body (key 7), as the ledger requires. Nothing balances it: the wallet only reads it.
 */
function withMetadata(metadata: string) {
  const hash = bytesToHex(blake2b(Buffer.from(metadata, "hex"), { dkLen: 32 }));
  const body = `a40081825820${INPUT}00018182${bytesOf(wasm.cip30Address(SITE))}1a003d0900021a00029810075820${hash}`;
  return `84${body}a0f5${metadata}`;
}

/** WebAssembly's summary of a site's transaction for the account, as readTx asks for it, and the window's text. */
function signWindow(txCbor: string, input: object = { tx_hash: INPUT, tx_index: 0, address: OWN, value: "10000000" }) {
  const cardano = wasm.CardanoAccount.fromPhrase(ME.phrase, 0);
  try {
    const request = JSON.stringify({
      network: "preprod",
      txCbor,
      keys: [{ role: 0, index: 0 }],
      inputs: [input],
      partialSign: false,
      stakeIndex: 0,
      governance: false,
    });
    const summary = JSON.parse(wasm.inspectDappTx(cardano, request)) as DappTxSummary;
    const html = renderToStaticMarkup(
      createElement(
        NetworkContext.Provider,
        { value: "preprod" },
        createElement(SignTx, { summary, partial: false, session: false, collateralSpent: false }),
      ),
    );
    return { summary, page: html.replace(/<[^>]+>/g, " ").replace(/&#x27;/g, "'").replace(/\s+/g, " ") };
  } finally {
    cardano.free();
  }
}

/** What Transaction details reads in the same bytes: the metadata, and whether the body commits to it. */
const detailOf = (txCbor: string) => JSON.parse(wasm.decodeTx(wasm.Network.Preprod, txCbor)) as TxDetail;

/** A metadatum's text, its lists' pieces joined (CIP-20 and Minswap split long text into 64-byte pieces). */
const joined = (m: TxMetadatum): string =>
  m.type === "text" ? m.text : m.type === "list" ? m.items.map(joined).join("") : "";

describe("the sign window and metadata beside a CIP-20 note (r51 x1a#3)", () => {
  it("says a site's transaction carries metadata when it has no note", () => {
    const tx = withMetadata(`a1${other}`);
    const detail = detailOf(tx);
    expect(detail.metadataHashMatches).toBe(true);
    expect(detail.metadata.map((m) => m.label)).toEqual(["1337"]);
    const { summary, page } = signWindow(tx);
    expect(summary.metadata).toBe(true);
    expect(summary.note).toBeNull();
    expect(page).toContain("It carries metadata, which anyone can read.");
  });

  it("names the metadata a site writes under another label beside its note", () => {
    const tx = withMetadata(`a2${note}${other}`);
    // The body commits to both labels: signing publishes the email and the order as well as the note.
    const detail = detailOf(tx);
    expect(detail.metadataHashMatches).toBe(true);
    expect(detail.metadata.map((m) => m.label)).toEqual(["674", "1337"]);
    const { summary, page } = signWindow(tx);
    expect(summary.complete).toBe(true);
    expect(summary.signs).toEqual(["0/0"]);
    expect(summary.metadata).toBe(true);
    expect(summary.note).toEqual([NOTE]);
    expect(page).toContain(`Its note, which anyone can read ${NOTE}`);
    // Beside the note, the window has to say there's more metadata than the note it shows.
    expect(page, "the window names no metadata beside the note").toMatch(/metadata/i);
  });

  it("names the data Minswap's real preprod swap writes beside its note: the sender's address and the order", () => {
    const fixture = JSON.parse(
      readFileSync(new URL("../../wasm/tests/fixtures/minswap-swap-preprod.json", import.meta.url), "utf8"),
    ) as { cbor: string; input: object };
    const detail = detailOf(fixture.cbor);
    expect(detail.metadataHashMatches).toBe(true);
    const label = detail.metadata.find((m) => m.label === "674")!.value;
    expect(label.type).toBe("map");
    const entries = label.type === "map" ? label.entries : [];
    expect(entries.map((e) => joined(e.key))).toEqual(["extraData", "msg"]);
    // Its extraData names the sender's whole address (the account's 0/0) and what the order asks for.
    const extra = joined(entries[0]!.value);
    expect(extra).toContain(OWN);
    expect(extra).toContain('"minimumAmountOut":{"$bigint":"902083681"}');
    const { summary, page } = signWindow(fixture.cbor, fixture.input);
    expect(summary.metadata).toBe(true);
    expect(summary.note).toEqual(["Minswap: Aggregator Market Order"]);
    expect(page).toContain("Its note, which anyone can read Minswap: Aggregator Market Order");
    expect(page, "the window names no metadata beside the note").toMatch(/metadata/i);
  });
});
