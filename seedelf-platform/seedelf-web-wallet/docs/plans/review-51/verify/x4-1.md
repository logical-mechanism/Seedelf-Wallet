# x4#1: verification

## Trace: confirmed, low

It holds as described. A new due withdraw reads the five-minute quiet rule once, at lovejoin.ts:6901. After that it waits on the network: made()'s Koios tx_info, rewardsIn's account_info, the WebAssembly build, then giveme.my's witness. Before its submit (7684 rememberSpent, 7692 submitTx) it checks only the hash and the unlock (7643, 7644), never sends() again. Three other senders run beside it without Lovejoin's turn: a page's Send or Make public (script-spend send, then pending writeAhead), Home's parts pump, and a site's submitTx. Each writes the records sends() reads, so a send inside that window is followed seconds later by the withdraw. That breaks privacy.md:442 and architecture.md:716 ("nor within 5 minutes of the wallet's own send"). My test drives a real public-account Send while giveme.my answers: both transactions go out in one run, 3 s apart. A 4-line re-check in a copy makes the same test pass. Graded low: a user, a site or an automatic part has to send inside a window of a few seconds per withdraw (up to about a minute under a Koios hold). The cost is a timing hint that pairs a box's way back with another of the wallet's transactions in the same or a nearby block. No money moves. The reverse order (a Send just after a withdraw) was never covered by the rule, so the protection lost is only half of it.

// Chunk 51, x4#1 (the trace verifier's test). A due box's withdraw reads Lovejoin's quiet rule (QUIET_AFTER_SEND_MS)
// once, before its Koios reads and giveme.my's witness (lovejoin.ts withdrawDueNow), and never again before its
// submit (withdrawOne). A Send that goes while giveme.my answers is followed seconds later by the withdraw: the very
// pairing the rule is for ("a withdraw minutes after a payment says both are one owner's"; privacy.md *Timing*:
// "nor within 5 minutes of the wallet's own send"). The right behaviour, asserted here: the withdraw isn't sent in
// that run, nothing is kept as on its way, its box keeps its due time, and it goes once the quiet after the Send is
// over. Fails today: the withdraw is submitted three seconds after the Send.
//
// The real WebAssembly, v2's 5 ₳ pool, a real Send from the public account, and fakes of Koios and giveme.my, set up
// as lovejoin-withdraw-submit.test.ts sets up its withdraws.
import { describe, expect, it, vi } from "vitest";

import { Collateral } from "../src/background/collateral";
import { LovejoinService, QUIET_AFTER_SEND_MS } from "../src/background/lovejoin";
import { lastSpentAt } from "../src/background/spent";
import { SESSION_UNLOCKED_AT } from "../src/background/wallet";
import { txIdOf } from "./fixtures/cbor";
import { busyFor, loadTestWasm, testBalances, vectors } from "./fakes";
import { ADA, ADA_POOL, ADA_ROWS, adaBox, madeByAdaMix, ownDatum } from "./v2-pools";

const PASSWORD = "correct horse battery";
const HOUR = 3_600_000;
const account = (words: number) =>
  vectors("cardano_account.json").find((v) => v.account === 0 && v.phrase.split(" ").length === words)!;
/** The 15-word phrase's receive address: someone else's. */
const THEIRS = account(15).preprod.receive_0 as string;
/** Each withdraw is built and measured in WebAssembly, and this test builds two, and a payment. */
const SLOW = { timeout: 120_000 };

type T = ReturnType<typeof testBalances>;

/** The Lovejoin record, with the 5 ₳ pool's due times as `due`. */
const kept = async (t: T) => {
  const s = (await t.store.get<{ pools?: Record<string, { due: number[] }>; withdrawing?: object }>("lovejoin.preprod"))!;
  return { ...s, due: s.pools?.[ADA]?.due ?? [] };
};

/**
 * As lovejoin-withdraw-submit.test.ts `due`: the 12-word phrase's wallet, unlocked an hour ago with its unlock's draws
 * made, one box of its own in the 5 ₳ pool, due an hour ago, and giveme.my's witness stood in for.
 */
async function due() {
  const t = testBalances({ pools: [ADA_POOL] });
  await t.wallet.create(account(12).phrase, PASSWORD);
  madeByAdaMix(t.koios, "e1".repeat(32));
  t.koios.addedToAccounts.push(...ADA_ROWS, adaBox("e1", await ownDatum(t.wallet), t.clock.now));
  const unlocked = t.clock.now - HOUR;
  await t.wallet.withKeys(() => t.session.set(SESSION_UNLOCKED_AT, unlocked));
  await t.store.set("lovejoin.preprod", { due: [], pools: { [ADA]: { due: [t.clock.now - HOUR] } }, unlock: unlocked });
  const wasm = loadTestWasm();
  t.collateral.answer = { status: 200, body: { witness: "a1008182" } };
  const lovejoin = new LovejoinService({
    ...t.deps,
    wasm: {
      ...wasm,
      finishLovejoinWithdraw: (request: string) => {
        const { txCbor } = JSON.parse(request) as { txCbor: string };
        return JSON.stringify({ txCbor, txHash: txIdOf(Uint8Array.from(Buffer.from(txCbor, "hex"))) });
      },
    } as typeof wasm,
    collateral: () => new Collateral("https://www.giveme.my/preprod/collateral/", t.collateral.fetch),
    store: t.store,
    pools: async () => [ADA_POOL],
  });
  return { t, lovejoin };
}

describe("r51 x4#1: a withdraw built while the user's Send goes", SLOW, () => {
  it("isn't sent seconds after the Send: it keeps its due time, and goes once the quiet after the Send is over", async () => {
    const { t, lovejoin } = await due();
    // The user reviews a payment from the public account. Nothing has been sent for an hour.
    const review = await t.send.build("preprod", [{ to: THEIRS, lovelace: "3000000", tokens: [] }]);
    expect(await t.wallet.sends()).toEqual({ sent: 0, forgotten: 0 });
    const before = (await kept(t)).due;
    const submits = t.koios.submitted.length;

    // The alarm's run begins (runs.ts `since`), passes the quiet check, reads the pool's reward account, builds the
    // withdraw and asks giveme.my to witness it. Three seconds in, while giveme.my answers, the user presses Send.
    const since = t.clock.now;
    const witness = t.collateral.fetch;
    let paid: number | undefined;
    t.collateral.fetch = async (url, init) => {
      t.clock.now += 3_000;
      paid = t.clock.now;
      // writeAhead stamps a Send's inputs on the device's clock (pending.ts:608), which in the worker is the services'
      // own (sw.ts `now: Date.now`). Here the services run on the test's clock, so the device's is set to it.
      const wall = vi.spyOn(Date, "now").mockReturnValue(paid);
      try {
        await t.send.submit("preprod", review.txHash);
      } finally {
        wall.mockRestore();
      }
      return witness(url, init);
    };
    const went = await lovejoin.withdrawDue("preprod", false, since);
    t.collateral.fetch = witness;

    // giveme.my was asked: the run had passed the quiet check. And the Send went meanwhile, the wallet's last send.
    expect(t.collateral.asked).toHaveLength(1);
    expect(await t.wallet.withKeys(() => lastSpentAt(t.session, t.clock.now))).toBe(paid);
    // Only the Send went out: no withdraw seconds after it.
    expect(t.koios.submitted.slice(submits).map(txIdOf)).toEqual([review.txHash]);
    expect(went).toEqual([]);
    // Nor is it kept as on its way: its box keeps its due time, for a later run.
    expect((await kept(t)).withdrawing).toBeUndefined();
    expect((await kept(t)).due).toHaveLength(before.length);

    // A minute on, still within the quiet after the Send, it doesn't go either.
    await busyFor(t, 60_000);
    await lovejoin.withdrawDue("preprod", false, t.clock.now);
    expect(t.koios.submitted).toHaveLength(submits + 1);

    // Once it's due and the quiet after the Send is over, it goes.
    const [at] = (await kept(t)).due;
    await busyFor(t, Math.max(at!, paid! + QUIET_AFTER_SEND_MS) - t.clock.now);
    const [back] = await lovejoin.withdrawDue("preprod", false, t.clock.now);
    expect(back?.kind).toBe("lovejoin-withdraw");
    expect(t.koios.submitted).toHaveLength(submits + 2);
    expect(t.clock.now - paid!).toBeGreaterThanOrEqual(QUIET_AFTER_SEND_MS);
  });
});
