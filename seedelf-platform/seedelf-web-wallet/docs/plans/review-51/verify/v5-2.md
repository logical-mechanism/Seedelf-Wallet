outf=$(mktemp); summary=()
for m in R51V52Demo.GuardedOk R51V52Demo.BareSorry R51V52Demo.GuardedSorry R51V52Demo.BareNative; do

  bash -c 'timeout "$1" lake build "$2"; exit $?' _ "$LIMIT" "$m" 2>&1 \

    | tee "$outf"
  rc=${PIPESTATUS[0]}
  undet=$(grep -cE '^warning: .*Undetermined' "$outf")

    0) summary+=("ok        $m") ;;
    undetermined) summary+=("UNDETERM  $m") ;;
    124) summary+=("TIMEOUT   $m") ;;
    137) summary+=("KILLED    $m") ;;
    *) summary+=("FAILED    $m (exit $rc)") ;;
  esac
done
rm -f "$outf"
echo; echo "== summary"; printf '%s\n' "${summary[@]}"
