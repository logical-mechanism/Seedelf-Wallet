# w2#2: verification

From lane W2, graded low by its finder.

## Trace: confirmed, low

The finding holds as described. The connector's guard at cip30.rs:1386 compares the payment part's 28 bytes (`as_hash()`) with the contract's script hash and never checks `is_script()`. Pallas 0.35.1's `as_hash()` returns the hash for a Key part as well as a Script part. So an enterprise KEY address whose key hash is the contract's script hash, which no key can spend (so what goes there is burned), gets `seedelf: Some("register")` (or `Some("none")` with no register). The window then calls it "Seedelf Wallet's contract" and adds "It pays a Seedelf…", or, with no register, the "anyone can take" warning. Transaction details (decode.rs:1370) checks `is_script()` and calls the same output a plain key address, so the two views disagree. I found no earlier guard, no caller that filters such an output, and no test that covers it: cip30_test.rs:1375 uses only the real script address. My test fails on TREE at exactly this assertion, and passes once the one-line `is_script()` guard goes in (checked in a copy). The harm is the window's words only: the amount and the full bech32 shown are true, nothing outside the window reads `paid[].seedelf`, and a hostile site gains nothing. Low.

//! Chunk 51, finding w2#2: the connector window names an output "Seedelf
//! Wallet's contract" by its payment part's 28 bytes alone. A KEY address whose
//! key hash is the contract's script hash matches too, though no key has that
//! hash: what's paid there is burned, not paid to a Seedelf. Transaction
//! details (`decode.rs`) checks `is_script()` and calls it a plain key address.
//! This asserts the right behaviour, so it fails while the bug is there.

use pallas_addresses::{
    Address, Network, ShelleyAddress, ShelleyDelegationPart, ShelleyPaymentPart,
};
use pallas_codec::utils::{Bytes, CborWrap, Nullable, Set};
use pallas_crypto::hash::Hash;
use pallas_primitives::{Fragment, PlutusData, TransactionInput, conway};
use seedelf_crypto::cardano::{CardanoAccount, Role};
use seedelf_wasm::cip30::{self, KeyPath, KoiosRow, TxRequest};
use seedelf_wasm::decode::decode_tx;

const PHRASE: &str =
    "abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon about";
const TX_A: &str = "1111111111111111111111111111111111111111111111111111111111111111";
/// A register datum: constructor 0 with the G1 generator and a test vector's public value, as cip30_test.rs has it.
const REGISTER_DATUM: &str = "d8799f583097f1d3a73197d7942695638c4fa9ac0fc3688c4f9774b905a14e3a3f171bac586c55e83ff97a1aeffb3af00adb22c6bb583082dcf46570656ca0d6fb143b8e7c2816b20cb1a6434ca4c8c95c624443c22c9e1d40ad0df5de088b19a4b44b685b8475ff";

fn account() -> CardanoAccount {
    CardanoAccount::from_phrase(PHRASE, 0).unwrap()
}

fn keys() -> Vec<KeyPath> {
    (0..3)
        .flat_map(|index| [KeyPath { role: 0, index }, KeyPath { role: 1, index }])
        .collect()
}

fn ours(role: Role, index: u32) -> Address {
    account().base_address(true, role, index).unwrap()
}

fn row(tx: &str, index: u64, address: &Address, lovelace: u64) -> KoiosRow {
    serde_json::from_value(serde_json::json!({
        "tx_hash": tx,
        "tx_index": index,
        "address": address.to_bech32().unwrap(),
        "value": lovelace.to_string(),
        "asset_list": [],
    }))
    .unwrap()
}

fn input(tx: &str, index: u64) -> TransactionInput {
    TransactionInput {
        transaction_id: Hash::new(hex::decode(tx).unwrap().try_into().unwrap()),
        index,
    }
}

fn out(address: &Address, lovelace: u64, datum: Option<&str>) -> conway::TransactionOutput {
    conway::PseudoTransactionOutput::PostAlonzo(conway::PostAlonzoTransactionOutput {
        address: Bytes::from(address.to_vec()),
        value: conway::Value::Coin(lovelace),
        datum_option: datum.map(|d| {
            conway::PseudoDatumOption::Data(CborWrap(
                PlutusData::decode_fragment(&hex::decode(d).unwrap()).unwrap(),
            ))
        }),
        script_ref: None,
    })
}

fn tx_hex(inputs: Vec<TransactionInput>, outputs: Vec<conway::TransactionOutput>) -> String {
    let body = conway::TransactionBody {
        inputs: Set::from(inputs),
        outputs,
        fee: 200_000,
        ttl: Some(99_999_999),
        certificates: None,
        withdrawals: None,
        auxiliary_data_hash: None,
        validity_interval_start: None,
        mint: None,
        script_data_hash: None,
        collateral: None,
        required_signers: None,
        network_id: None,
        collateral_return: None,
        total_collateral: None,
        reference_inputs: None,
        voting_procedures: None,
        proposal_procedures: None,
        treasury_value: None,
        donation: None,
    };
    let tx = conway::Tx {
        transaction_body: body,
        transaction_witness_set: conway::WitnessSet {
            vkeywitness: None,
            native_script: None,
            bootstrap_witness: None,
            plutus_v1_script: None,
            plutus_data: None,
            redeemer: None,
            plutus_v2_script: None,
            plutus_v3_script: None,
        },
        success: true,
        auxiliary_data: Nullable::Null,
    };
    hex::encode(tx.encode_fragment().unwrap())
}

#[test]
fn a_key_address_with_the_contracts_hash_is_not_seedelf_wallets_contract() {
    let contract_hash = seedelf_core::constants::get_config(1, true)
        .unwrap()
        .contract
        .wallet_contract_hash;
    // The contract itself: its hash as a SCRIPT payment part.
    let contract = seedelf_core::address::wallet_contract(true, contract_hash);
    // The same 28 bytes as a KEY payment part, enterprise, on preprod (header
    // 0x60): an address no key spends, so what goes there is gone for good.
    let burn = Address::Shelley(ShelleyAddress::new(
        Network::Testnet,
        ShelleyPaymentPart::Key(Hash::new(contract_hash)),
        ShelleyDelegationPart::Null,
    ));
    assert_eq!(burn.to_vec()[0], 0x60, "an enterprise key address on a test network");
    assert_ne!(burn.to_vec(), contract.to_vec());

    // The account's 10 ₳ pays 2 ₳ there under a register, 1.5 ₳ there with no
    // datum, 2 ₳ into the real contract under a register, and its change.
    let rows = vec![row(TX_A, 0, &ours(Role::Receive, 0), 10_000_000)];
    let tx = tx_hex(
        vec![input(TX_A, 0)],
        vec![
            out(&burn, 2_000_000, Some(REGISTER_DATUM)),
            out(&burn, 1_500_000, None),
            out(&contract, 2_000_000, Some(REGISTER_DATUM)),
            out(&ours(Role::Change, 0), 4_300_000, None),
        ],
    );

    // Transaction details reads the key address as what it is: a key's, not
    // the contract's (decode.rs checks is_script()). Holds today.
    let detail = decode_tx("preprod", &tx).unwrap();
    assert_eq!(detail.outputs[0].address.payment.as_deref(), Some("key"));
    assert!(!detail.outputs[0].address.seedelf);
    assert!(!detail.outputs[1].address.seedelf);
    assert!(detail.outputs[2].address.seedelf);

    let request = TxRequest {
        network: "preprod".into(),
        tx_cbor: tx,
        keys: keys(),
        inputs: rows,
        partial_sign: false,
        stake_index: 0,
        stake_deposit: None,
        governance: false,
    };
    let summary = cip30::inspect_tx(&account(), &request).unwrap();
    assert_eq!(summary.paid.len(), 3, "the change is the account's own");
    assert_eq!(summary.paid[0].address, burn.to_bech32().unwrap());
    // Not a script, so no contract holds it.
    assert!(!summary.paid[0].script);
    assert!(!summary.paid[1].script);
    // The real contract output still reads as a Seedelf payment.
    assert!(summary.paid[2].script);
    assert_eq!(summary.paid[2].seedelf.as_deref(), Some("register"));
    // The window must not call a key address Seedelf Wallet's contract:
    // "register" shows "Seedelf Wallet's contract" and "It pays a Seedelf…";
    // "none" warns "…without a register: whatever goes there, anyone can take."
    assert_eq!(
        summary.paid[0].seedelf, None,
        "a key address under the register datum is labelled Seedelf Wallet's contract"
    );
    assert_eq!(
        summary.paid[1].seedelf, None,
        "a key address with no datum gets the contract's no-register warning"
    );
}
