//! Chunk 51's v3#2 (1.8.1's release review left it to the owner: "a CI change"). A pull
//! request that changes only a file a test reads must run the check that reads it.
//! `web-wallet.yml`'s `changes` job decides that with one awk program, `covered`.
//! `lovejoin_proven_bytes_test.rs` reads four `.hex` files in `seedelf-verification/`,
//! outside `seedelf-platform/`, and `covered` matches none of them. So a PR into main that
//! edits only one gets `platform=false`, every job skipped and *Web Wallet required* green,
//! and the tie between the proofs' bytes and the wallet's goes red only on main's run.

use std::fs;
use std::io::Write;
use std::path::{Path, PathBuf};
use std::process::{Command, Stdio};

/// The workflow (the filter covers this path itself, so the scan below checks it too).
const WORKFLOW: &str = include_str!("../../../../.github/workflows/web-wallet.yml");

/// What `lovejoin_proven_bytes_test.rs` reads, so the scan can't pass by finding nothing.
const PROVEN: [&str; 4] = [
    "seedelf-verification/Lovejoin/Template/Scripts/mix_logic.hex",
    "seedelf-verification/Lovejoin/Template/Scripts/mix_box.hex",
    "seedelf-verification/Lovejoin/Template/Scripts/one_shot_mint.hex",
    "seedelf-verification/Lovejoin/Scripts/reference_holder.hex",
];

/// `seedelf-platform/`, and the repository around it.
fn roots() -> (PathBuf, PathBuf) {
    let platform = Path::new(env!("CARGO_MANIFEST_DIR"))
        .join("../..")
        .canonicalize()
        .unwrap();
    let repo = platform.parent().unwrap().to_path_buf();
    (platform, repo)
}

/// The `changes` step's awk program, as bash's single quotes hand it to awk.
fn covered() -> &'static str {
    let start = WORKFLOW
        .find("covered='")
        .expect("web-wallet.yml's changes step has no covered='…'")
        + "covered='".len();
    let end = start + WORKFLOW[start..].find('\'').unwrap();
    &WORKFLOW[start..end]
}

/// Whether `changes` says a PR into main changing only `path` touches what the check covers:
/// with one file listed and `changed_files` 1, the awk program alone decides.
fn runs_the_check(path: &str) -> bool {
    let mut awk = Command::new("awk")
        .arg(covered())
        .stdin(Stdio::piped())
        .spawn()
        .expect("awk");
    let mut stdin = awk.stdin.take().unwrap();
    stdin.write_all(format!("{path}\n").as_bytes()).unwrap();
    drop(stdin);
    awk.wait().unwrap().success()
}

/// Every `.rs` file under `dir`, but build output and dependencies.
fn sources(dir: &Path, out: &mut Vec<PathBuf>) {
    for entry in fs::read_dir(dir).unwrap() {
        let path = entry.unwrap().path();
        let name = path.file_name().unwrap().to_string_lossy().into_owned();
        if path.is_dir() {
            if !matches!(
                name.as_str(),
                "target" | "node_modules" | "dist" | "pkg" | ".git"
            ) {
                sources(&path, out);
            }
        } else if name.ends_with(".rs") {
            out.push(path);
        }
    }
}

/// Each file outside `seedelf-platform/` that a source in it includes, repository-relative.
fn read_from_outside() -> Vec<String> {
    let (platform, repo) = roots();
    let mut files = Vec::new();
    sources(&platform, &mut files);
    let mut outside = Vec::new();
    for file in files {
        let text = fs::read_to_string(&file).unwrap();
        for mac in ["include_str", "include_bytes"] {
            for (at, _) in text.match_indices(mac) {
                let Some(rest) = text[at + mac.len()..].strip_prefix("!(") else {
                    continue;
                };
                let Some(lit) = rest.trim_start().strip_prefix('"') else {
                    continue;
                };
                let lit = &lit[..lit.find('"').unwrap()];
                let Ok(target) = file.parent().unwrap().join(lit).canonicalize() else {
                    continue;
                };
                if !target.starts_with(&platform) {
                    let rel = target.strip_prefix(&repo).unwrap();
                    outside.push(rel.to_string_lossy().replace('\\', "/"));
                }
            }
        }
    }
    outside.sort();
    outside.dedup();
    outside
}

#[test]
fn a_pr_changing_only_a_file_a_test_reads_runs_the_check() {
    // The harness reads the filter right: code and wasm/README.md run the check, the
    // workflow does, Markdown elsewhere doesn't.
    assert!(runs_the_check(
        "seedelf-platform/seedelf-core/src/lovejoin/templates.rs"
    ));
    assert!(runs_the_check(
        "seedelf-platform/seedelf-web-wallet/wasm/README.md"
    ));
    assert!(runs_the_check(".github/workflows/web-wallet.yml"));
    assert!(!runs_the_check(
        "seedelf-platform/seedelf-web-wallet/docs/backlog.md"
    ));

    let outside = read_from_outside();
    for file in PROVEN {
        assert!(
            outside.iter().any(|f| f == file),
            "the scan missed {file}, which lovejoin_proven_bytes_test.rs reads"
        );
    }
    let skipped: Vec<&String> = outside.iter().filter(|f| !runs_the_check(f)).collect();
    assert!(
        skipped.is_empty(),
        "web-wallet.yml's changes job skips the check for a PR into main that changes only \
         one of {skipped:?}, which a test reads"
    );
}
