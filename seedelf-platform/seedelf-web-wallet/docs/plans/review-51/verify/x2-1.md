//! Chunk 51, x2#1: a pool's rows are every UTxO whose payment credential is
//! its `mix_box` hash M, as Koios's `credential_utxos` lists them. It takes
//! bare 28-byte hashes, a key's or a script's alike (the wallet asks it for
//! both), so a UTxO at an enterprise KEY address whose key hash is M comes
//! back beside the boxes, with `payment_cred` = M. Anyone can pay one: no key
//! hashes to M, so its ADA is burnt. The contract never takes it as a box
//! (`input_at_script` wants `Script(M)`), and the ledger asks no script and no
//! redeemer of a key-locked input. So the wallet must never read it as a box:
//! never count it among the others a fan-out needs, and never draw it into a
//! mix, which the wallet's own measure then refuses (uplc's phase one: a
//! redeemer the transaction doesn't need).

    ProtocolParameters::from_koios(&rows[0]).unwrap()
}

/// v2's pool of 5 ₳ boxes of ADA on preprod, verified as the wallet verifies
/// one (`lovejoin_pools_test.rs`'s `v2_ada`): where a return's ADA goes.
fn v2_ada() -> Protocol {
    let doc: Json = serde_json::from_str(include_str!(
        "../../seedelf-web-wallet/wasm/tests/fixtures/lovejoin-v2-ada-pool-preprod.json"
    ))
    .unwrap();
    let row = |v: &Json| serde_json::from_value::<UtxoResponse>(v.clone()).unwrap();
    let reference = eval::resolve_row(&row(&doc["reference"])).unwrap();
    let scripts: Vec<Resolved> = doc["scripts"]
        .as_array()
        .unwrap()
        .iter()
        .map(|s| {
            let bytes = hex::decode(s["reference_script"]["bytes"].as_str().unwrap()).unwrap();
            eval::resolve_reference(&row(s), 3, &bytes).unwrap()
        })
        .collect();
    let seed: [u8; 32] = hex::decode(doc["seed"]["txHash"].as_str().unwrap())
        .unwrap()
        .try_into()
        .unwrap();
    let pool = lovejoin::verify_pool(
        true,
        &reference,
        (seed, doc["seed"]["txIndex"].as_u64().unwrap()),
        &scripts,
    )
    .unwrap();
    assert_eq!(pool.version, Version::V2);
    assert_eq!(pool.denom, 5_000_000);
    pool
}

fn random_scalar() -> Scalar {
    loop {
        let s = Scalar::random(&mut OsRng);
        if !bool::from(s.is_zero()) {
            return s;
        }
    }
}

/// A box's `{a, b}`: valid G1 points, neither the identity, `a ≠ b`; owned by
/// `sk` (`b = sk·a`) when given.
fn points(sk: Option<&Scalar>) -> ([u8; 48], [u8; 48]) {
    let a = G1Projective::generator() * random_scalar();
    let b = a * sk.copied().unwrap_or_else(random_scalar);
    (crypto::point_to_bytes(&a), crypto::point_to_bytes(&b))
}

/// The enterprise KEY address whose key hash is `pool`'s `mix_box` script
/// hash: header 0x60 on preprod, 0x61 on mainnet.
fn twin(pool: &Protocol) -> Address {
    let network = if pool.network_flag {
        Network::Testnet
    } else {
        Network::Mainnet
    };
    ShelleyAddress::new(
        network,
        ShelleyPaymentPart::Key(Hash::new(pool.mix_box_hash)),
        ShelleyDelegationPart::Null,
    )
    .into()
}

/// A row as Koios's `credential_utxos` for M lists it: exactly the box value
/// at `address`, a box's inline datum, and `payment_cred` M (the key hash of
/// the twin is M, as the script hash of `mix_box` is).
fn row(pool: &Protocol, address: &Address, first: u8, (a, b): ([u8; 48], [u8; 48])) -> UtxoResponse {
    serde_json::from_value(json!({
        "tx_hash": hex::encode([first; 32]), "tx_index": 0,
        "address": address.to_bech32().unwrap(),
        "value": pool.denom.to_string(), "stake_address": null,
        "payment_cred": hex::encode(pool.mix_box_hash), "epoch_no": 0,
        "block_height": 0, "block_time": 0, "datum_hash": null,
        "inline_datum": { "bytes": hex::encode(lovejoin::mix_datum(&a, &b)), "value": null },
        "reference_script": null, "asset_list": [], "is_spent": false,
    }))
    .unwrap()
}

/// A session's base address on preprod, and ADA alone at it.
fn session() -> Address {
    let mut raw = vec![0x00];
    raw.extend([0x51; 28]);
    raw.extend([0x52; 28]);
    Address::from_bytes(&raw).unwrap()
}

fn coin(first: u8, lovelace: u64) -> Coin {
    Coin {
        utxo: Resolved {
            tx_hash: [first; 32],
            index: 0,
            output: Output::new(session(), lovelace)
                .build_babbage_raw()
                .unwrap()
                .encode_fragment()
                .unwrap(),
        },
        lovelace,
    }
}

fn payer() -> Payer {
    Payer {
        fee: coin(0xf1, 50_000_000),
        collateral: Some(coin(0xcc, 5_000_000)),
        address: session(),
        signers: 1,
    }
}

fn a_key_address_under_a_pools_mix_box_hash_is_never_read_as_a_box() {
    let params = params();
    let mut wrong: Vec<String> = Vec::new();

    // 1. The row itself: in v2's 5 ₳ pool, and the classic pools of both networks.
    for pool in [
        v2_ada(),
        Protocol::of(true).unwrap(),
        Protocol::of(false).unwrap(),
    ] {
        let header = |a: &Address| a.to_vec()[0] >> 4;
        assert_eq!(header(&pool.mix_box_address()), 0x7, "mix_box: enterprise script");
        assert_eq!(header(&twin(&pool)), 0x6, "the twin: enterprise key");
        let real = row(&pool, &pool.mix_box_address(), 0x31, points(None));
        assert!(
            PoolBox::from_row(&real, &pool).is_some(),
            "a real box is read as one"
        );
        let fake = row(&pool, &twin(&pool), 0x32, points(None));
        if let Some(b) = PoolBox::from_row(&fake, &pool) {
            wrong.push(format!(

                 of the pool's: PoolBox {{ a: {}…, utxo output's address header {:#04x} }}",
                pool.denom / 1_000_000,
                twin(&pool).to_bech32().unwrap(),
                hex::encode(pool.mix_box_hash),
                &hex::encode(b.a)[..16],
                b.utxo.output.get(4).copied().unwrap_or_default(),
            ));
        }
    }

    // 2. A mix of the boxes the wallet reads from v2's 5 ₳ pool's rows: three
    // real ones and the twin. Every box it reads must be one a mix can take.
    let pool = v2_ada();
    let mut rows: Vec<UtxoResponse> = (0..3u8)

        .collect();
    rows.push(row(&pool, &twin(&pool), 0x4f, points(None)));
    let boxes: Vec<PoolBox> = rows
        .iter()
        .filter_map(|r| PoolBox::from_row(r, &pool))
        .collect();
    if let Err(e) = lovejoin::mix(&params, &pool, &boxes, &payer(), &mut Rewards::none()) {
        wrong.push(format!(
            "[mix] the wallet read {} boxes from 3 real rows and the twin, and a mix of them \
             fails in the wallet's own measure: {e:#}",
            boxes.len()
        ));
    }

    // 3. Mixing again one box of the wallet's, one wave three wide, where the
    // pool's others are one real box and the twin: it needs two others, and
    // the pool has one, so it's short (PoolShort), never a chain that draws

    let sk = random_scalar();

        row(&pool, &pool.mix_box_address(), 0x50, points(Some(&sk))),
        row(&pool, &pool.mix_box_address(), 0x51, points(None)),
        row(&pool, &twin(&pool), 0x52, points(None)),
    ];
    let (ours, others): (Vec<PoolBox>, Vec<PoolBox>) = rows
        .iter()
        .filter_map(|r| PoolBox::from_row(r, &pool))
        .partition(|b| b.is_owned(&sk));
    assert_eq!(ours.len(), 1, "the wallet's own box is read as its");
    match lovejoin::again_at(&params, &pool, &payer(), &ours, 1, 3, &others, &mut Rewards::none()) {

            others.len()
        )),
        Ok(_) => wrong.push(format!(
            "[fan-out] a chain built from {} others where only 1 is a box",
            others.len()
        )),
    }

    assert!(wrong.is_empty(), "\n{}", wrong.join("\n\n"));
}

## Impact: confirmed, medium

The impact holds at medium, and it reaches more users than the finder claimed. The data layer's committed SQL gathers pool rows by `address.payment_cred` with no `has_script` check, and the wallet's `credOf`/`keysOf` doesn't check the credential's kind either. So mainnet's default route serves a key-M fake to every Seedelf Wallet user, not only on Koios-only, preprod or data-layer-down paths. A stranger who burns 5 ₳ per fake can make the default Lovejoin return come back directly for every user of the 5 ₳ pool, which makes the session link that privacy.md:404 describes. The effect lasts until a release changes the check. It isn't high: no funds are locked, stranded or misdirected, the fake is never signed for (phase one refuses it locally), and the money lost is only the fees of mix sessions that fail.
