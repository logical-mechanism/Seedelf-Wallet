# NOT a TREE test: a read-only scan, run with `python3 -I v5_1_coverage_check.py <TREE>/seedelf-verification` from the scratch folder.
"""Read-only check for chunk 51's v5#1: which mint line counts the policy's Blaster
theorems cover, what PolicyShapes' family asks of them, and what CLAIMS.md and the README
say about it. Takes the seedelf-verification folder as its one argument; writes nothing."""
import itertools
import re
import sys
from pathlib import Path

root = Path(sys.argv[1])


def read(rel):
    return (root / rel).read_text(encoding="utf-8")


def compositions(n):
    out = []
    for cuts in itertools.product([0, 1], repeat=n - 1):
        parts, run = [], 1
        for c in cuts:
            if c:
                parts.append(run)
                run = 1
            else:
                run += 1
        parts.append(run)
        out.append(tuple(parts))
    return set(out)


# 1. The Blaster theorems over multi-line mints, and the line counts they cover.
files = ["Seedelf/PolicyLinesAnyTx.lean", "Seedelf/PolicyLinesFourAnyTx.lean", "Seedelf/PolicyLinesFiveAnyTx.lean"]
thm = re.compile(r"^theorem (policy_fails_(?:no_lines|lines_(\d+))) :\n(.*?)(?=^theorem |^--|^#|\Z)", re.M | re.S)
layouts = {}
for f in files:
    for m in thm.finditer(read(f)):
        name, digits, body = m.group(1), m.group(2), m.group(3)
        assert body.rstrip().endswith("by blaster"), (f, name)
        layouts[name] = tuple(int(d) for d in digits) if digits else ()
counts = sorted({sum(l) for l in layouts.values()})
print(f"Blaster theorems over a B-tagged mint of other than one line: {len(layouts)}")
print(f"line counts they cover: {counts}")
for n in range(2, 6):
    have = {l for l in layouts.values() if sum(l) == n}
    print(f"  {n} lines: {len(have)} of {len(compositions(n))} layouts, all present: {have == compositions(n)}")
print(f"  6 lines: 0 of {len(compositions(6))} layouts; 7 lines: 0 of {len(compositions(7))}")

# 2. PolicyShapes' third family: what the cover hands those modules.
shapes = read("Seedelf/PolicyShapes.lean")
fam = shapes[shapes.index("def policyFamilies"):shapes.index("theorem policy_every_context_cover")]
print("PolicyShapes family 3 has `lineCount lines ≠ 1`:", "lineCount lines ≠ 1" in fam)
print("PolicyShapes family 3 bounds the count above:", bool(re.search(r"lineCount lines (≤|<) ", fam)))

# 3. CLAIMS.md: does it name the policy's Blaster line bound, as it names the wallet's?
claims = read("CLAIMS.md")
for i, line in enumerate(claims.splitlines(), 1):
    if "Blaster" in line and ("128" in line or "PolicyLines" in line or "cross-check" in line):
        for frag in re.findall(r"[^.]*(?:128|PolicyLines\*AnyTx|cross-check by another route)[^.]*\.", line):
            print(f"CLAIMS.md:{i}: {frag.strip()}")
bound = re.compile(r"\b(five|six|2 to 5|5 lines|up to five|lines? (?:or|and) more)\b", re.I)
print("CLAIMS.md names a mint-line bound:", [m.group(0) for m in bound.finditer(claims)] or "no")
print("CLAIMS.md names the wallet's Blaster bound (128):", "128" in claims)

# 4. The README: where it does state the bound, its 'One property', and the :202 anchor.
readme = read("README.md").splitlines()
for i, line in enumerate(readme, 1):
    if re.search(r"2 to 5 lines|arrangement of up to five|One property is proved|By Blaster alone", line):
        print(f"README.md:{i}: {line[:150]}")
heads = [l.lstrip("# ").strip() for l in readme if l.startswith("#")]
print("README heading 'By Blaster alone' exists:", "By Blaster alone" in heads)
print("README heading 'Proved in the kernel, not yet by Blaster' exists:", "Proved in the kernel, not yet by Blaster" in heads)
