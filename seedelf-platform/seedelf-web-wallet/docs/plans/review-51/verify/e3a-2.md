// Chunk 51, finding e3a#2: the connector's pick of a private session's
// collateral (dapp.ts sessionCollateral) is the first UTxO of exactly 5 ₳ with

/** A UTxO at session 0's account, as Koios lists it. */
function atSession(tx_hash: string, tx_index: number, value: string): KoiosUtxo {
  return {
    ...(sessionSwap.utxo as unknown as KoiosUtxo),
    tx_hash,
    tx_index,
    value,
    payment_cred: sessionSwap.keyHash,
    stake_address: null,
    block_height: 5_000_000,
    asset_list: [],
    is_spent: false,
  } as KoiosUtxo;
}

/** The connector over a session service, as dapp.test.ts' `privately`: giveme.my and the Seedelf proof stood in for. */
function privately(t: T) {

const uint = (n: number) => (n < 24 ? n.toString(16).padStart(2, "0") : `18${n.toString(16).padStart(2, "0")}`);

/** A site's transaction spending `inputs` (`txhash#index`) and paying 4 ₳ to someone else; nothing balances it. */
function siteTx(inputs: string[]) {
  const ins = `8${inputs.length}${inputs.map((o) => `825820${o.slice(0, 64)}${uint(Number(o.slice(65)))}`).join("")}`;
  const outputs = `81825839${loadTestWasm().cip30Address(THEIRS)}1a003d0900`;
  return `84a300${ins}01${outputs}021a00029810a0f5f6`;
}

/** A CIP-30 `TransactionUnspentOutput`'s outpoint: `[[txhash, index], output]`, an index under 24. */
const outpointOf = (hex: string) => `${hex.slice(8, 72)}#${parseInt(hex.slice(72, 74), 16)}`;

/** A stranger's transaction id below any funding's: Koios lists its outputs first. */
const STRANGER = "00".repeat(32);

/**
 * A site connected to session 0, funded with 15 ₳ and its 5 ₳ collateral
 * (`out.txHash#1`), whose account Koios lists as `rows` gives it: the
 * recorded swap's UTxO, the funding's 5 ₳ and the stranger's 5 ₳.
 */
async function connected(t: T, dapp: DappService, stranger: Partial<KoiosUtxo>, strangerFirst: boolean) {
  const s = site();
  const enabling = dapp.call(s, "enable", []);
  await until(() => dapp.approvals().length === 1);
  const out = await dapp.privateBuild(dapp.approvals()[0]!.id, "15000000", []);
  expect(await dapp.answer(dapp.approvals()[0]!.id, true, PASSWORD, { txHash: out.txHash })).toEqual({});
  const theirs = { ...atSession(STRANGER, 0, "5000000"), ...stranger } as KoiosUtxo;
  const swap = atSession(sessionSwap.utxo.tx_hash, sessionSwap.utxo.tx_index, sessionSwap.utxo.value);
  const own = atSession(out.txHash, 1, "5000000");
  t.koios.addedToAccounts.push(...(strangerFirst ? [theirs, swap, own] : [swap, own, theirs]));
  expect(await enabling).toBe(true);
  return { s, own: `${out.txHash}#1`, theirs: `${STRANGER}#0` };
}

/** What the connector tells the site, and whether a transaction spending the funding's 5 ₳ is flagged. */
async function whatTheSiteSees(dapp: DappService, s: DappSession, own: string) {
  const collateral = ((await dapp.call(s, "getCollateral", [])) as string[] | null)?.map(outpointOf) ?? null;
  const utxos = ((await dapp.call(s, "getUtxos", [])) as string[]).map(outpointOf);
  const signing = dapp.call(s, "signTx", [siteTx([own]), false]);
  await until(() => dapp.approvals().length === 1);
  const approval = dapp.approvals()[0]!;
  await dapp.answer(approval.id, false);
  await expect(signing).rejects.toMatchObject({ failure: { code: TxSignError.UserDeclined } });
  return { collateral, utxos, approval };
}
