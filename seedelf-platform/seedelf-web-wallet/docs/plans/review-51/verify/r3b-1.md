Nothing pins that order: no test sends a deep swap to the dry run, and deep.test.mjs covers three other bindings but not `checkOrderCancels`. Grade: who = nobody. The cost would be a trap that locks the wallet on a fresh instance (runs.ts:57-61, handlers.ts:50), with no funds lost and unlocking as the way out. That is "a missing check with no reachable harm today", which is low on the brief's scale. The finder's grade stands.

//! Chunk 51, r3b#1: `orders::check_cancels` decodes the swap Minswap built
//! (bytes from outside the wallet) with `MultiEraTx::decode` and no depth
//! check of its own. Pallas decodes plutus data by recursion, so a datum a
//! few thousand levels deep overflows WebAssembly's stack and traps the
//! module (the launch review's #17, 1.6.0's 314-wasm#0). Today the wallet's
//! one caller is safe only because `cip30::inspect` runs `check_nesting` on
//! the same bytes first (sessions.ts `inspect`, then `cancelsChecked`).
//!
//! The right behaviour: the check refuses such a swap itself, before
//! decoding it, wherever the deep datum sits (in the witness set, or inline
//! in an output, inside tag 24's bytes, which Pallas decodes too), and an
//! honest swap still checks out. The deep cases run on a thread with a large
//! stack, so that natively they show what `check_cancels` returns rather than
//! overflow the test's own stack.

use pallas_addresses::{
    Address, Network, ShelleyAddress, ShelleyDelegationPart, ShelleyPaymentPart,
};
use pallas_crypto::hash::{Hash, Hasher};
use pallas_txbuilder::{BuildConway, Input, Output, StagingTransaction};
use seedelf_core::cbor::{MAX_DEPTH, within_depth};
use seedelf_core::orders::{self, Credential, Protocol, Session, SwapCheck};
use seedelf_koios::koios::{ProtocolParameters, UtxoResponse};
use serde_json::{Value, json};

/// Levels of nesting: past the few thousand that overflowed WebAssembly's
/// stack, and about 4 KB, well within a transaction's 16 KiB.
const DEEP: usize = 4_000;

fn params() -> ProtocolParameters {
    let rows: Value =
        serde_json::from_str(include_str!("fixtures/epoch_params_mainnet.json")).unwrap();
    let row = if rows.is_array() { &rows[0] } else { &rows };
    ProtocolParameters::from_koios(row).unwrap()
}

/// A Koios row from a fixture's UTxO (as orders_test.rs reads them).
fn row(u: &Value) -> UtxoResponse {
    let address = Address::from_bech32(u["address"].as_str().unwrap()).unwrap();
    let payment_cred = match &address {
        Address::Shelley(s) => hex::encode(s.payment().as_hash()),
        _ => String::new(),
    };
    serde_json::from_value(json!({
        "tx_hash": u["tx_hash"],
        "tx_index": u["tx_index"],
        "address": u["address"],
        "value": u["value"],
        "stake_address": null,
        "payment_cred": payment_cred,
        "epoch_no": 0,
        "block_height": 0,
        "block_time": 0,
        "datum_hash": u.get("datum_hash").cloned().unwrap_or(Value::Null),
        "inline_datum": u.get("inline_datum").cloned().unwrap_or(Value::Null),
        "reference_script": u.get("reference_script").cloned().unwrap_or(Value::Null),
        "asset_list": u.get("asset_list").cloned().unwrap_or(json!([])),
        "is_spent": false,
    }))
    .unwrap()
}

/// `n` levels of one-item lists around a 0: plutus data Pallas decodes one
/// call deeper a level.
fn nested(n: usize) -> Vec<u8> {
    let mut v = vec![0x81; n];
    v.push(0x00);
    v
}

/// Minswap V2's real mainnet order (tests/fixtures/order_cancels.json), its
/// owner as the session.
struct Setup {
    session: Session,
    address: Address,
    order: UtxoResponse,
    datum: Vec<u8>,
    references: Vec<UtxoResponse>,
    params: ProtocolParameters,
}

impl Setup {
    fn minswap_v2() -> Self {
        let fixtures: Vec<Value> =
            serde_json::from_str(include_str!("fixtures/order_cancels.json")).unwrap();
        let f = fixtures
            .iter()
            .find(|f| f["protocol"] == "MinswapV2" && f["network"] == "mainnet")
            .unwrap();
        let protocol: Protocol = serde_json::from_value(json!("MinswapV2")).unwrap();
        let datum = hex::decode(f["datum"].as_str().unwrap()).unwrap();
        let to = orders::read(protocol, &datum).unwrap().pays[0];
        let (Credential::Key(payment), Some(Credential::Key(stake))) = (to.payment, to.stake)
        else {
            panic!("Minswap V2's order doesn't pay a base address of keys");
        };
        let session = Session { payment, stake };
        Setup {
            session,
            address: Address::Shelley(ShelleyAddress::new(
                Network::Mainnet,
                ShelleyPaymentPart::key_hash(payment),
                ShelleyDelegationPart::key_hash(stake),
            )),
            order: row(&f["order"]),
            datum,
            references: vec![row(&f["reference"])],
            params: params(),
        }
    }

    /// A swap as Minswap builds one: the session's change at output 0, the
    /// order (its datum inline) at output 1, and, when `deep` is given, an
    /// output at the session's address under a datum nested `DEEP` levels:
    /// carried in the witness set for its hash, or inline.
    fn swap(&self, deep: Option<Deep>) -> Vec<u8> {
        let mut order = Output::new(
            Address::from_bech32(&self.order.address).unwrap(),
            self.order.value.parse().unwrap(),
        );
        for a in self.order.asset_list.iter().flatten() {
            let policy: [u8; 28] = hex::decode(&a.policy_id).unwrap().try_into().unwrap();
            order = order
                .add_asset(
                    Hash::new(policy),
                    hex::decode(&a.asset_name).unwrap(),
                    a.quantity.parse().unwrap(),
                )
                .unwrap();
        }
        let mut tx = StagingTransaction::new()
            .input(Input::new(Hash::new([0x77; 32]), 0))
            .output(Output::new(self.address.clone(), 3_000_000))
            .output(order.set_inline_datum(self.datum.clone()));
        match deep {
            None => {}
            Some(Deep::Witness) => {
                let datum = nested(DEEP);
                tx = tx
                    .output(
                        Output::new(self.address.clone(), 2_000_000)
                            .set_datum_hash(Hasher::<256>::hash(&datum)),
                    )
                    .datum(datum);
            }
            Some(Deep::Inline) => {
                tx = tx.output(
                    Output::new(self.address.clone(), 2_000_000).set_inline_datum(nested(DEEP)),
                );
            }
        }
        tx.fee(200_000).build_conway_raw().unwrap().tx_bytes.0
    }

    /// `check_cancels` on the swap's order at output 1.
    fn check(&self, tx: &[u8]) -> anyhow::Result<Vec<orders::CancelCheck>> {
        orders::check_cancels(&SwapCheck {
            params: &self.params,
            network_flag: false,
            session: self.session,
            address: &self.address,
            tx_cbor: tx,
            orders: &[1],
            references: &self.references,
        })
    }
}

#[derive(Clone, Copy, Debug)]
enum Deep {
    Witness,
    Inline,
}

#[test]
fn a_swap_nested_too_deep_is_refused_by_the_order_check_itself() {
    let run = std::thread::Builder::new()
        .stack_size(256 << 20)
        .spawn(|| {
            let s = Setup::minswap_v2();

            // The honest swap checks out: its order's cancel builds and runs.
            let honest = s.swap(None);
            assert!(within_depth(&honest, MAX_DEPTH));
            let checks = s.check(&honest).expect("an honest swap is read");
            assert_eq!(checks.len(), 1);
            assert_eq!(checks[0].why, None, "{:?}", checks[0]);
            assert!(checks[0].fee.is_some(), "{:?}", checks[0]);

            // The same swap with a datum `DEEP` levels down. Core's own walk
            // already calls the witness set's too deep (a tag-24 inline datum
            // it doesn't open; cip30's `nests_within` does).
            let mut wrong = Vec::new();
            for deep in [Deep::Witness, Deep::Inline] {
                let tx = s.swap(Some(deep));
                eprintln!(
                    "{deep:?}: {} bytes, core's within_depth(tx, {MAX_DEPTH}) = {}",
                    tx.len(),
                    within_depth(&tx, MAX_DEPTH)
                );
                match s.check(&tx) {
                    Err(e) if e.to_string().contains("nested") => {
                        eprintln!("{deep:?}: refused: {e}");
                    }
                    other => {
                        let got = format!("{other:?}");
                        eprintln!("{deep:?}: check_cancels returned {got}");
                        wrong.push(format!("{deep:?}: {got}"));
                    }
                }
            }
            assert!(
                wrong.is_empty(),
                "check_cancels decoded a swap nested {DEEP} levels deep instead of refusing it: {wrong:#?}"
            );
        })
        .unwrap();
    if let Err(panic) = run.join() {
        std::panic::resume_unwind(panic);
    }
}
