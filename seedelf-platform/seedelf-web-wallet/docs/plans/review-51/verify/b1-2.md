// Chunk 51, b1#2. wasm/build.sh builds the store's WebAssembly with whatever Rust rustup picks, and never checks
// or says which. rustup's order is +toolchain, then RUSTUP_TOOLCHAIN, then a directory override, then the nearest
// rust-toolchain.toml from the working folder up, then the default. So a release shell that sets RUSTUP_TOOLCHAIN
// builds the module with another Rust. build.sh's last lines, which are what the release record copies, name only
// the C compiler. part 2's second build runs in the same terminal, gets the same variable and agrees.
//
// The reachable case is narrower than "any installed toolchain". Here 1.96.x and nightly have no wasm32 target, and
// cargo refuses 1.90.0 and 1.93.0 because uplc@1.1.23 needs rustc 1.94.1. So it takes a toolchain other than the pin
// that has the target and rustc >= 1.94.1. `stable` has the target, and after a `rustup update` it is a later Rust.
// That is the case below: RUSTUP_TOOLCHAIN=stable, with stable at 1.99.0.
//
// This runs a copy of TREE's build.sh, in a scratch mirror of the workspace that holds the real Cargo.lock and
// rust-toolchain.toml. Stand-ins replace rustc, cargo, rustup, wasm-bindgen, clang and llvm-ar, and pick the
// toolchain in rustup's order. Nothing is compiled and nothing under TREE is written. The right behaviour: build.sh
// refuses a Rust other than the pinned one, builds with the pinned one anyway, or names the one it built with.

import { spawnSync } from "node:child_process";
import { chmodSync, copyFileSync, mkdirSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { fileURLToPath } from "node:url";

import { afterAll, expect, it } from "vitest";

const wasm = fileURLToPath(new URL("../../wasm/", import.meta.url));
const platform = fileURLToPath(new URL("../../../", import.meta.url));

const lock = readFileSync(join(platform, "Cargo.lock"), "utf8");
const pinned = /^channel\s*=\s*"([^"]+)"/m.exec(readFileSync(join(platform, "rust-toolchain.toml"), "utf8"))?.[1];
const bindgen = /^name = "wasm-bindgen"\nversion = "([^"]+)"/m.exec(lock)?.[1];
/** What `stable` resolves to once a `rustup update` has moved it past the pin. */
const LATER_STABLE = "1.99.0";

/** A bash script as written, raw: its `${…}` are spelled `@{…}` here so the template literal leaves them alone. */
function sh(strings: TemplateStringsArray): string {
  return strings.raw.join("").replaceAll("@{", "${");
}

const RESOLVE = sh`# rustup's choice: +toolchain, RUSTUP_TOOLCHAIN, the nearest rust-toolchain.toml up from the working
# folder, the default. (A directory override would come before the file; none is set here.)
toolchain() {
  if [[ "@{1:-}" == +* ]]; then echo "@{1#+}"; return; fi
  if [[ -n "@{RUSTUP_TOOLCHAIN:-}" ]]; then echo "$RUSTUP_TOOLCHAIN"; return; fi
  local d="$PWD"
  while :; do
    if [[ -f "$d/rust-toolchain.toml" ]]; then sed -n 's/^channel = "\(.*\)"$/\1/p' "$d/rust-toolchain.toml"; return; fi
    [[ "$d" == / ]] && break
    d="$(dirname "$d")"
  done
  echo "$FAKE_DEFAULT"
}
release_of() { case "$1" in stable) echo "$FAKE_STABLE" ;; *) echo "$1" ;; esac; }
# Each release its own commit, as each official toolchain has.
hash_of() { printf '%040d' "$(printf '%s' "$1" | cksum | cut -d' ' -f1)"; }
`;

const RUSTC = sh`#!/usr/bin/env bash
. "$(dirname "$0")/resolve.sh"
tc="$(toolchain "$@")"; [[ "@{1:-}" == +* ]] && shift
r="$(release_of "$tc")"; h="$(hash_of "$r")"
a=" $* "
if [[ "$a" == *" --print sysroot "* || "$a" == *" --print=sysroot "* ]]; then echo "$FAKE_ROOT/toolchains/$tc-x86_64-unknown-linux-gnu"; exit 0; fi
if [[ "$a" == *" -vV "* || "$a" == *" -Vv "* || ( "$a" == *" -V "* || "$a" == *" --version "* ) && ( "$a" == *" -v "* || "$a" == *" --verbose "* ) ]]; then
  printf 'rustc %s (%s 2026-01-01)\nbinary: rustc\ncommit-hash: %s\ncommit-date: 2026-01-01\nhost: x86_64-unknown-linux-gnu\nrelease: %s\nLLVM version: 21.1.8\n' "$r" "@{h:0:9}" "$h" "$r"
  exit 0
fi
if [[ "$a" == *" -V "* || "$a" == *" --version "* ]]; then echo "rustc $r (@{h:0:9} 2026-01-01)"; exit 0; fi
echo "stand-in rustc: not made for: $*" >&2; exit 2
`;

const CARGO = sh`#!/usr/bin/env bash
. "$(dirname "$0")/resolve.sh"
tc="$(toolchain "$@")"; [[ "@{1:-}" == +* ]] && shift
r="$(release_of "$tc")"; h="$(hash_of "$r")"
case "@{1:-}" in
  -V|--version) echo "cargo $r (@{h:0:9} 2026-01-01)"; exit 0 ;;
  build) shift ;;
  *) echo "stand-in cargo: not made for: $*" >&2; exit 2 ;;
esac
profile=dev; target=""
while [[ $# -gt 0 ]]; do
  case "$1" in
    --profile) profile="$2"; shift 2 ;;
    --profile=*) profile="@{1#*=}"; shift ;;
    --target) target="$2"; shift 2 ;;
    --target=*) target="@{1#*=}"; shift ;;
    --manifest-path) shift 2 ;;
    *) shift ;;
  esac
done
out="@{CARGO_TARGET_DIR:-$FAKE_WORKSPACE/target}/$target/$profile"
mkdir -p "$out"
# A module whose std names its compiler's commit, as a real one does (/rustc/<commit-hash>/library/...).
printf '\0asm\1\0\0\0/rustc/%s/library/core/src/panicking.rs' "$h" > "$out/seedelf_wasm.wasm"
echo "$r $h" >> "$FAKE_LOG"
echo "   Compiling seedelf-wasm v0.7.0 (/seedelf-platform/seedelf-web-wallet/wasm)" >&2
echo "    Finished \`$profile\` profile [optimized] target(s) in 0.01s" >&2
`;

const RUSTUP = sh`#!/usr/bin/env bash
. "$(dirname "$0")/resolve.sh"
if [[ "@{1:-}" == show && "@{2:-}" == active-toolchain ]]; then echo "$(toolchain)-x86_64-unknown-linux-gnu"; exit 0; fi
if [[ "@{1:-}" == run && $# -ge 3 ]]; then tc="$2"; shift 2; RUSTUP_TOOLCHAIN="$tc" exec "$@"; fi
echo "stand-in rustup: not made for: $*" >&2; exit 2
`;

const WASM_BINDGEN = sh`#!/usr/bin/env bash
if [[ "@{1:-}" == --version || "@{1:-}" == -V ]]; then echo "wasm-bindgen $FAKE_BINDGEN"; exit 0; fi
out=""; in=""
while [[ $# -gt 0 ]]; do
  case "$1" in
    --out-dir) out="$2"; shift 2 ;;
    --target|--out-name) shift 2 ;;
    --*) shift ;;
    *) in="$1"; shift ;;
  esac
done
mkdir -p "$out" && cp "$in" "$out/seedelf_wasm_bg.wasm" && : > "$out/seedelf_wasm.js"
`;

const roots: string[] = [];
afterAll(() => {
  for (const root of roots) rmSync(root, { recursive: true, force: true });
});

interface Run {
  status: number | null;
  output: string;
  /** The Rust the stand-in cargo built the module with, if it ran. */
  builtWith?: { release: string; hash: string };
}

/** TREE's build.sh, run as `npm run build:wasm` runs it (from extension/), in a fresh mirror, with `extra` in its shell. */
function build(extra: Record<string, string>): Run {
  const root = mkdtempSync(join(tmpdir(), "r51-b1-2-"));
  roots.push(root);
  const workspace = join(root, "seedelf-platform");
  const here = join(workspace, "seedelf-web-wallet", "wasm");
  const extension = join(workspace, "seedelf-web-wallet", "extension");
  const bin = join(root, "bin");
  const home = join(root, "home");
  for (const dir of [here, extension, bin, home]) mkdirSync(dir, { recursive: true });
  copyFileSync(join(platform, "Cargo.lock"), join(workspace, "Cargo.lock"));
  copyFileSync(join(platform, "rust-toolchain.toml"), join(workspace, "rust-toolchain.toml"));
  copyFileSync(join(wasm, "build.sh"), join(here, "build.sh"));
  chmodSync(join(here, "build.sh"), 0o755);
  writeFileSync(join(bin, "resolve.sh"), RESOLVE);
  const tools: Record<string, string> = {
    rustc: RUSTC,
    cargo: CARGO,
    rustup: RUSTUP,
    "wasm-bindgen": WASM_BINDGEN,
    clang: '#!/usr/bin/env bash\necho "Ubuntu clang version 21.1.8 (stand-in)"\n',
    "llvm-ar": "#!/usr/bin/env bash\nexit 0\n",
  };
  for (const [name, text] of Object.entries(tools)) {
    writeFileSync(join(bin, name), text);
    chmodSync(join(bin, name), 0o755);
  }
  const log = join(root, "built-with.log");
  writeFileSync(log, "");
  // A shell of its own: no CARGO_TARGET_DIR, CARGO_HOME or RUSTUP_* from this one, and no real Rust on PATH.
  const env: Record<string, string> = {
    PATH: `${bin}:/usr/local/bin:/usr/bin:/bin`,
    HOME: home,
    LANG: "C.UTF-8",
    FAKE_ROOT: root,
    FAKE_WORKSPACE: workspace,
    FAKE_LOG: log,
    FAKE_BINDGEN: bindgen ?? "",
    FAKE_DEFAULT: "stable",
    FAKE_STABLE: LATER_STABLE,
    ...extra,
  };
  const run = spawnSync("bash", [join(here, "build.sh")], { cwd: extension, env, encoding: "utf8", timeout: 60_000 });
  const last = readFileSync(log, "utf8").trim().split("\n").filter(Boolean).at(-1);
  const [release, hash] = last?.split(" ") ?? [];
  return {
    status: run.status,
    output: `${run.stdout ?? ""}${run.stderr ?? ""}`,
    builtWith: release && hash ? { release, hash } : undefined,
  };
}

it("reads the pin and wasm-bindgen's version from TREE", () => {
  expect(pinned).toBe("1.98.1");
  expect(bindgen).toMatch(/^\d+\.\d+\.\d+$/);
});

it("the mirror: from extension/, with nothing overriding it, the pinned Rust builds the module", () => {
  const run = build({});
  expect(run.status, run.output).toBe(0);
  expect(run.builtWith?.release, run.output).toBe(pinned);
});

it("a release shell's RUSTUP_TOOLCHAIN: build.sh refuses that Rust, builds with the pinned one, or names it", () => {
  const run = build({ RUSTUP_TOOLCHAIN: "stable" });
  const built = run.builtWith;
  const keptThePin = run.status === 0 && built?.release === pinned;
  const refused = run.status !== 0 && (run.output.includes(LATER_STABLE) || run.output.includes(pinned ?? "?"));
  const namedIt =
    run.status === 0 && built !== undefined && (run.output.includes(built.release) || run.output.includes(built.hash));
  expect(
    keptThePin || refused || namedIt,
    `build.sh exited ${run.status} with the module built by Rust ${built?.release ?? "none"}, while ` +
      `rust-toolchain.toml pins ${pinned}, and its output never names the Rust it used:\n${run.output}`,
  ).toBe(true);
});
