Grade (low): the user reaches it by an ordinary action with default Settings, on mainnet whenever v2's 5 ₳ pool is shown, over its floor and the spare ADA pays for a box. It costs no money, and Send sends exactly the reviewed chain. What goes wrong is which transaction the review lets the user inspect, and its label. There are ways out: turn the switch off (a direct return is one transaction, opened correctly), use Bring everything back (it opens the same session's deposit), or after Send use the Lovejoin page's chain list. It's not a nit, because it changes what can be inspected, not only wording. It's not medium, because no money, figure, stuck state or privacy link is involved.

// Chunk 51's x1b#1: a site session's Bring it back review through Lovejoin opens Transaction details on its chain's
// last transaction, the return, under the plain words, where every other chain review opens the chain's first, the
// deposit, and names it (6bd936f4; flows.md, "A chain's review shows the transaction the money goes in through").
//
// The worker's side is pinned already (tests/chain-entry.test.ts): a session's return through Lovejoin keeps its chain
// under SESSION_BACK, the summary's `txHash` is the chain's last ("back") and `lovejoin.entry` its first ("deposit"),
// and the view opens either by hash. What no test pins is which of the two a screen hands its button. Vitest has no
// DOM, and a review is a state only an awaited build reaches, so `back` starts from a seed in its useState, as
// screens-wiring-review-fixes.test.ts seeds it. TxDetailButton is the real one, its props recorded on the way in.
import { createElement, type ReactElement } from "react";
import { renderToStaticMarkup } from "react-dom/server";
import { beforeEach, describe, expect, it, vi } from "vitest";

const seam = vi.hoisted(() => ({
  /** The useState right after a run of states whose initial values match `after`, in order, starts as `value`. */
  seeds: [] as Array<{ after: Array<(v: unknown) => boolean>; value: unknown; used?: true }>,
  /** Every useState's initial value, in the order called. */
  initials: [] as unknown[],
  /** Every TxDetailButton made, with the props that say what it opens and what it reads. */
  buttons: [] as Array<{ txHash: string; label?: string; testId?: string }>,
}));

vi.mock("react", async (original) => {
  const react = await original<typeof import("react")>();
  const useState = (initial?: unknown) => {
    const before = seam.initials;
    seam.initials = [...before, initial];
    for (const seed of seam.seeds) {
      if (seed.used || initial !== undefined || before.length < seed.after.length) continue;
      const run = before.slice(before.length - seed.after.length);
      if (!seed.after.every((want, i) => want(run[i]))) continue;
      seed.used = true;
      return react.useState(seed.value);
    }
    return react.useState(initial);
  };
  return { ...react, useState };
});
// The worker never answers: nothing on a review's render asks it anything, and a press isn't made here.
vi.mock("../src/ui/background", async (original) => ({
  ...(await original<typeof import("../src/ui/background")>()),
  call: () => new Promise(() => undefined),
}));
vi.mock("../src/ui/components/TxDetail", async (original) => {
  const real = await original<typeof import("../src/ui/components/TxDetail")>();
  return {
    ...real,
    TxDetailButton: (props: Parameters<typeof real.TxDetailButton>[0]) => {
      seam.buttons.push({ txHash: props.txHash, label: props.label, testId: props.testId });
      return real.TxDetailButton(props);
    },
  };

function render(element: ReactElement): string {
  return renderToStaticMarkup(
    createElement(
      NetworkContext.Provider,
      { value: "preprod" },
      createElement(PreferencesContext.Provider, { value: { prefs: DEFAULT_PREFERENCES, loaded: true, set: async () => undefined } }, element),
    ),
  );
}

/** What a person reads of markup. */
const words = (markup: string) =>
  markup
    .replace(/<[^>]+>/g, " ")
    .replaceAll("&#x27;", "'")
    .replaceAll("&amp;", "&")
    .replace(/\s+/g, " ")
    .trim();

/** The site session's page, its return's review built as `back`: its states start with which part shows, then the return. */
function siteReview(back: unknown): string {
  seam.seeds.push({ after: [(v) => v === "main"], value: back });
  return render(
    createElement(SiteSession, { session, seedelf, reading: false, onRefresh: noop, onBack: noop, onPending: noop, onDisconnected: noop } as never),
  );
}

/** A swap's page, the same return's review built: its states start with the session, its orders and a step's review. */
function swapReview(back: unknown): string {
  seam.seeds.push({ after: [(v) => v === session, (v) => v === undefined, (v) => v === undefined], value: back });
  return render(createElement(Session, { session, reading: false, onRefresh: noop, onBack: noop, onChanged: noop } as never));
}

/** What the review's button opens (`txHash`) and what it says (`label`, none for "Transaction details"). */
const opens = (testId: string) => {
  const made = seam.buttons.filter((b) => b.testId === testId);
  expect(made.length, `${testId} made`).toBeGreaterThan(0);
  const { txHash, label } = made.at(-1)!;
  return { txHash, label };
};

beforeEach(() => {
  seam.seeds = [];
  seam.initials = [];
  seam.buttons = [];
});

describe("which transaction a session's return review opens (chunk 51, x1b#1)", () => {
  it("a site's return through Lovejoin opens its chain's first transaction, the deposit, and names it", () => {
    const markup = siteReview(through);
    expect(markup).toContain('data-testid="site-back-review"');
    expect(opens("site-back-tx")).toEqual({ txHash: DEPOSIT, label: t("tx.entry.deposit") });
    const said = words(markup);
    expect(said).toContain("The deposit's transaction");
    expect(said).not.toContain(t("tx.detailsButton"));
  });

  it("as the same return reviewed from a swap's page does (6bd936f4): the harness reads both the same way", () => {
    const markup = swapReview(through);
    expect(markup).toContain('data-testid="session-back-review"');
    expect(opens("session-back-tx")).toEqual({ txHash: DEPOSIT, label: t("tx.entry.deposit") });
  });

  it("a site's return that comes back directly is one transaction: it opens that one, under the plain words", () => {
    const markup = siteReview(direct);
    expect(markup).toContain('data-testid="site-back-review"');
    expect(opens("site-back-tx")).toEqual({ txHash: RETURN, label: undefined });
    expect(words(markup)).toContain(t("tx.detailsButton"));
  });
});
