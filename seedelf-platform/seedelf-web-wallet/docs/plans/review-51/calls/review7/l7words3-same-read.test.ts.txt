// l7words#3 (trace check): privacy.md:458 says of Bring every box back now "the pool read it makes after it stops is the
// same read every wallet makes", and :462 says the boxes the retired pools' run leaves "get theirs from a pool read at
// the next run, the same read every wallet makes, not 2 hours on". These tests hold the code to those words: a read
// every wallet makes sends Koios no tx_info about a box of this wallet's. Helpers copied from r6-when-it-asks.test.ts
// (A) and lovejoin-round2-retire.test.ts (B).
import { describe, expect, it } from "vitest";

import { txInputs } from "../src/background/cbor";
import { Collateral } from "../src/background/collateral";
import type { KoiosUtxo } from "../src/background/koios";
import { BRING_RUN_MS, LovejoinService, type LovejoinDeps, type PoolRef } from "../src/background/lovejoin";
import { NETWORKS } from "../src/networks";
import { account, AGREES, CHAINS, HOUR, PASSWORD, POOL, withSession, type Tested } from "./chain-fixtures";
import { txIdOf } from "./fixtures/cbor";
import { busyFor, loadTestWasm, testBalances } from "./fakes";
import { ADA_POOL, ADA_ROWS, adaBox, ownDatum, retiredAlready } from "./v2-pools";

const wasm = loadTestWasm();
const hash = (h: string) => (h.length === 64 ? h : h.repeat(32));
const CLASSIC_BOX = NETWORKS.preprod.lovejoin!.mixBox;

/** Every tx_info request Koios was asked, the hashes each asked for. */
const txInfoAsked = (t: { koios: { calls: Array<{ path: string; body: any }> } }) =>
  t.koios.calls.filter((c) => c.path === "tx_info").map((c) => [...(c.body._tx_hashes as string[])].sort());
const asking = async (t: { store: Tested["store"] }) => (await t.store.get<{ asking?: Record<string, number> }>("lovejoin.preprod"))?.asking;
const recheckOf = async (t: { store: Tested["store"] }) => (await t.store.get<{ recheck?: number }>("lovejoin.preprod"))?.recheck;
const withdrawn = (t: { collateral: { asked: string[] } }) => t.collateral.asked.flatMap((tx) => txInputs(Uint8Array.from(Buffer.from(tx, "hex"))));

// ---------------------------------------------------------------------------
// A: privacy.md:458, the read Bring every box back now owes 2 hours after a Stop
// ---------------------------------------------------------------------------

const adaOwned = async (t: Tested, tx: string) => adaBox(hash(tx), await ownDatum(t.wallet), t.clock.now, 5, 0);

function witnessed(t: Tested) {
  t.collateral.answer = { status: 200, body: { witness: "a1008182" } };
  return new LovejoinService({
    ...t.deps,
    wasm: {
      ...wasm,
      finishLovejoinWithdraw: (request: string) => {
        const { txCbor } = JSON.parse(request) as { txCbor: string };
        return JSON.stringify({ txCbor, txHash: txIdOf(Uint8Array.from(Buffer.from(txCbor, "hex"))) });
      },
    } as typeof wasm,
    collateral: () => new Collateral("https://www.giveme.my/preprod/collateral/", t.collateral.fetch),
    store: t.store,
    pools: async () => [ADA_POOL],
  });
}

function slowFirstWitness(t: Tested) {
  const fetch = t.collateral.fetch;
  let first = true;
  t.collateral.fetch = async (url, init) => {
    if (first) {
      first = false;
      t.clock.now += BRING_RUN_MS;
    }
    return fetch(url, init);
  };
}

describe("l7words#3 A: privacy.md:458, the pool read Bring every box back now makes after it stops", CHAINS, () => {
  it("is the same read every wallet makes: it asks Koios nothing about a box of the wallet's", async () => {
    const { t } = await withSession("40000000");
    await retiredAlready(t.store);
    // After a restore, the Lovejoin page's read asked Koios what made U and V, and Koios didn't answer.
    const [U, V] = [hash("e9"), hash("d9")];
    t.koios.addedToAccounts.push(await adaOwned(t, U), await adaOwned(t, V));
    await t.lovejoin.status("preprod");
    const waiting = Object.keys((await asking(t)) ?? {}).sort();
    expect(waiting).toEqual([U, V].sort());
    const lovejoin = witnessed(t);
    await busyFor(t, 60_000);
    await lovejoin.bringAllBack("preprod");
    const before = txInfoAsked(t).length;
    slowFirstWitness(t);
    await lovejoin.withdrawDue("preprod");
    const sent = withdrawn(t);
    expect(sent).toHaveLength(1);
    await lovejoin.stopBringingAllBack("preprod");
    expect(await recheckOf(t)).toBeGreaterThan(t.clock.now);
    for (const box of sent) t.koios.spent.add(box);
    for (let minute = 0; minute < 3; minute++) {
      await busyFor(t, 60_000);
      await lovejoin.withdrawDue("preprod");
    }
    // Nothing due and no unlock: the only read left is the one owed 2 hours after the Stop.
    expect(txInfoAsked(t).slice(before)).toEqual([]);
    await busyFor(t, 2 * HOUR);
    await lovejoin.withdrawDue("preprod");
    // privacy.md:458: "the pool read it makes after it stops is the same read every wallet makes".
    expect(txInfoAsked(t).slice(before)).toEqual([]);
  });
});

// ---------------------------------------------------------------------------
// B: privacy.md:462, the read retireOnce owes at the next run once the retired pools' run stops
// ---------------------------------------------------------------------------

type T = ReturnType<typeof testBalances>;

function pooled(t: T, pools: PoolRef[] = [ADA_POOL], extra: Partial<LovejoinDeps> = {}): LovejoinService {
  return new LovejoinService({
    ...t.deps,
    wasm: {
      ...wasm,
      finishLovejoinWithdraw: (request: string) => {
        const { txCbor } = JSON.parse(request) as { txCbor: string };
        return JSON.stringify({ txCbor, txHash: txIdOf(Uint8Array.from(Buffer.from(txCbor, "hex"))) });
      },
    } as typeof wasm,
    collateral: () => new Collateral("https://www.giveme.my/preprod/collateral/", (u, i) => t.collateral.fetch(u, i)),
    store: t.store,
    pools: async () => pools,
    random: () => 0.5,
    ...extra,
  });
}

async function wallet(): Promise<T> {
  const t = testBalances();
  await t.wallet.create(account(12).phrase, PASSWORD);
  t.koios.evaluation = AGREES;
  t.collateral.answer = { status: 200, body: { witness: "a1008182" } };
  return t;
}

async function classicBox(t: T, tx: string, hours = 5, index = 0): Promise<KoiosUtxo> {
  const h = hash(tx);
  t.koios.txSpends.set(h, [{ payment_addr: { bech32: POOL[0]!.address, cred: CLASSIC_BOX } }]);
  return {
    ...POOL[0]!,
    tx_hash: h,
    tx_index: index,
    block_time: Math.floor(t.clock.now / 1000) - hours * 3600,
    inline_datum: { bytes: await ownDatum(t.wallet), value: {} },
  } as KoiosUtxo;
}

const at = (u: KoiosUtxo) => `${u.tx_hash}#${u.tx_index}`;
function landed(t: T, only: string[]): void {
  const gone = new Set(only);
  const kept = t.koios.addedToAccounts.filter((u) => !gone.has(at(u)));
  t.koios.addedToAccounts.splice(0, t.koios.addedToAccounts.length, ...kept);
}

function givemeTakes(t: T, ok: number, then: { status: number; body: unknown }): () => void {
  const yes = { status: 200, body: { witness: "a1008182" } };
  let calls = 0;
  const base = t.collateral.fetch;
  t.collateral.fetch = async (u, i) => {
    t.collateral.answer = calls++ < ok ? yes : then;
    return base(u, i);
  };
  return () => {
    t.collateral.fetch = base;
    t.collateral.answer = yes;
  };
}

/**
 * lovejoin-round2-retire's heldBy17: two classic boxes a deposit made, 1.7's count of them not mixed yet kept, so
 * retireOnce owes a read at once. With `waiting`, also U, a classic box no record names, which a 1.7 pool read asked
 * Koios about and Koios hasn't answered for (`asking`), listed after them.
 */
async function heldBy17(t: T, waiting: boolean, dueOnly = false): Promise<{ boxes: KoiosUtxo[]; u?: KoiosUtxo }> {
  const boxes = [];
  for (const tx of ["c1", "c2"]) {
    const box = await classicBox(t, tx, 10);
    t.koios.txSpends.set(box.tx_hash, [{ payment_addr: { bech32: "addr_test1", cred: "ab".repeat(28) } }]);
    boxes.push(box);
  }
  const u = waiting ? await classicBox(t, "e7", 10) : undefined;
  if (u) t.koios.txSpends.delete(u.tx_hash);
  await t.store.set("lovejoin.preprod", {
    // dueOnly: U never asked of, a 1.7 due time its own (one someone else's mix moved, in a 1.7 wallet's steady state).
    due: u && dueOnly ? [t.clock.now + 5 * HOUR] : [],
    chains: [],
    notMixed: 2,
    origins: Object.fromEntries(boxes.map((b) => [b.tx_hash, { mixed: false, seen: t.clock.now - HOUR }])),
    ...(u && !dueOnly ? { asking: { [u.tx_hash]: t.clock.now - HOUR } } : {}),
  });
  t.koios.addedToAccounts.push(...POOL, ...ADA_ROWS, ...boxes, ...(u ? [u] : []));
  return { boxes, ...(u ? { u } : {}) };
}

async function retiredRunStoppedThenNextRun(waiting: boolean, dueOnly = false) {
  const t = await wallet();
  const lj = pooled(t);
  const { boxes, u } = await heldBy17(t, waiting, dueOnly);
  const restore = givemeTakes(t, 1, { status: 400, body: { detail: "Transaction Fails Validation" } });
  const before = txInfoAsked(t).length;
  // The unlock's run: retireOnce asks the retired pools' run, which sends one withdraw, then the user stops it.
  expect(await lj.withdrawDue("preprod", true)).toHaveLength(1);
  const run = txInfoAsked(t).slice(before);
  await lj.stopBringingAllBack("preprod");
  // The one giveme.my witnessed (the boxes after it were asked of and refused, so they're still there).
  const took = txInputs(Uint8Array.from(Buffer.from(t.collateral.asked[0]!, "hex")));
  expect(took).toEqual([at(boxes[0]!)]);
  expect(t.koios.addedToAccounts.some((x) => u && at(x) === at(u))).toBe(!!u);
  landed(t, took);
  restore();
  // retireOnce's read is owed at once, and Stop never puts it later: it comes at the next run, not 2 hours on.
  expect(await recheckOf(t)).toBeLessThanOrEqual(t.clock.now);
  await busyFor(t, 60_000);
  await lj.withdrawDue("preprod");
  return { run, next: txInfoAsked(t).slice(before + run.length), recheck: await recheckOf(t), u };
}

describe("l7words#3 B: privacy.md:462, the read at the next run once the retired pools' run stops", { timeout: 300_000 }, () => {
  it("control: a wallet with no box Koios is asked about: the read asks Koios nothing", async () => {
    const { run, next, recheck } = await retiredRunStoppedThenNextRun(false);
    expect(run).toEqual([]);
    expect(recheck).toBeUndefined();
    expect(next).toEqual([]);
  });

  it("is the same read every wallet makes: it asks Koios nothing about a box of the wallet's", async () => {
    const { run, next, recheck } = await retiredRunStoppedThenNextRun(true);
    // The run's own reads ask nothing (privacy.md:253, the policy: "their own pool reads don't ask it either").
    expect(run).toEqual([]);
    expect(recheck).toBeUndefined();
    // privacy.md:462: "get theirs from a pool read at the next run, the same read every wallet makes, not 2 hours on".
    expect(next).toEqual([]);
  });

  it("the same, U never asked of before (a 1.7 due time its own): the read asks Koios nothing about it", async () => {
    const { run, next, recheck } = await retiredRunStoppedThenNextRun(true, true);
    expect(run).toEqual([]);
    expect(recheck).toBeUndefined();
    expect(next).toEqual([]);
  });
});
